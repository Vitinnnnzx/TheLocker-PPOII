CREATE TABLE atleta_time (
    id SERIAL PRIMARY KEY,
    atleta_id INTEGER NOT NULL,
    time_id INTEGER NOT NULL,
    data_entrada DATE,
    data_saida DATE,

    FOREIGN KEY (atleta_id)
        REFERENCES atleta(id)
        ON DELETE CASCADE,

    FOREIGN KEY (time_id)
        REFERENCES time(id)
        ON DELETE CASCADE
);
CREATE INDEX atleta_time_time_id_idx ON atleta_time (time_id);
CREATE INDEX atleta_time_atleta_id_idx ON atleta_time (atleta_id);
